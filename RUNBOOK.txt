================================================================================
RUNBOOK — ARP Flood Detection using P4/BMv2
5 switches (s1, s2, s3, s4 edges; s11 central), 60 hosts (15 per edge)
================================================================================

--------------------------------------------------------------------------------
0. PRE-FLIGHT (do once)
--------------------------------------------------------------------------------
cd ~/Desktop/Arp_Flood_Detection-using-P4-Bmv2-main

# Compile the P4 program → build/arp_monitor.json
./compile.sh

# Sanity-check the compiled header size (must print 45)
python3 -c "import json; j=json.load(open('build/arp_monitor.json')); \
ht=next(x for x in j['header_types'] if x['name']=='cpu_header_t'); \
print('cpu_header_t:', sum(f[1] for f in ht['fields'])//8, 'bytes')"

# Sanity-check the controller format string (must print 45)
python3 -c "import struct; print('CPU_SIZE =', struct.calcsize('!BB H 4s 6s 4s H IIIII BI'))"

# Clean any stale Mininet / BMv2 state
sudo mn -c
sudo pkill -9 -f simple_switch 2>/dev/null || true

# Clear stale runtime config files from previous runs (optional)
rm -f switch_config/*_runtime.txt

--------------------------------------------------------------------------------
1. TERMINAL 1 — START THE TOPOLOGY (Mininet + 5 BMv2 switches)
--------------------------------------------------------------------------------
cd ~/Desktop/Arp_Flood_Detection-using-P4-Bmv2-main
sudo python3 topology/topology.py

# Wait for:
#   [TOPO] NETWORK READY (5 switches, 60 hosts)
#   mininet>
#
# At the mininet> prompt, smoke-test connectivity:
#   mininet> h1  ping -c 2 10.1.2.1      (same edge)
#   mininet> h1  ping -c 2 10.2.1.1      (edge1 -> edge2)
#   mininet> h1  ping -c 2 10.3.1.1      (edge1 -> edge3)
#   mininet> h1  ping -c 2 10.4.1.1      (edge1 -> edge4)
#   mininet> h46 ping -c 2 10.1.1.1      (reverse)
#   mininet> pingall                     (60*59 = 3540 pairs, ~30 s)
#
# Expected: *** Results: 0% dropped (3540/3540 received)
#
# Leave this terminal running. Do NOT close it.

--------------------------------------------------------------------------------
2. TERMINAL 2 — START THE CONTROLLER (monitoring + optional blocking)
--------------------------------------------------------------------------------
cd ~/Desktop/Arp_Flood_Detection-using-P4-Bmv2-main

sudo python3 controller/controller.py \
  --interfaces s1-cpu1 s2-cpu1 s3-cpu1 s4-cpu1 s11-cpu1 \
  --label normal

# Expected startup:
#   [CSV] /home/abhishek/Desktop/Arp_Flood_Detection-using-P4-Bmv2-main/dataset/arp_dataset_new.csv
#   [SNIFF] Capturing on s1-cpu1 ...
#   [SNIFF] Capturing on s2-cpu1 ...
#   [SNIFF] Capturing on s3-cpu1 ...
#   [SNIFF] Capturing on s4-cpu1 ...
#   [SNIFF] Capturing on s11-cpu1 ...
#   [CTRL] PID=<n>  Running. Ctrl+C to stop.
#
# One [WIN] line and one [TOP_SRC] line are printed every ~1 second.

--------------------------------------------------------------------------------
3. TERMINAL 1 (mininet>) — GENERATE ARP TRAFFIC FOR NORMAL DATA
--------------------------------------------------------------------------------
# quick traffic (small amount)
mininet> h1 ping -c 20 10.2.1.1

# heavy traffic (large amount, all hosts)
mininet> pingall

# targeted cross-edge traffic
mininet> h1  ping -c 20 10.4.15.1
mininet> h30 ping -c 20 10.3.1.1

--------------------------------------------------------------------------------
4. TOGGLE THE LABEL BETWEEN NORMAL AND ATTACK
--------------------------------------------------------------------------------
# From ANY terminal (must run as root because the controller runs as root):
sudo pkill -USR1 -f "controller/controller.py"

# Each SIGUSR1 toggles the CSV 'label' column:
#   normal (0)  ->  attack (1)  ->  normal (0)  ->  ...
#
# Convention:
#   1. Generate normal traffic, keep label = 0.
#   2. Send SIGUSR1 (label becomes 1).
#   3. Generate attack traffic.
#   4. Send SIGUSR1 (label becomes 0).
#   5. Repeat.

--------------------------------------------------------------------------------
5. GENERATE ATTACK TRAFFIC (from mininet>)
--------------------------------------------------------------------------------
# ARP request flood from one host (1000 packets, no wait)
mininet> h1 arping -I h1-eth0 -c 1000 10.2.1.1

# ARP flood at max rate (very high pps, run for ~10 s then Ctrl+C)
mininet> h1 hping3 --flood -1 10.2.1.1

# Gratuitous ARP flood
mininet> h1 arping -I h1-eth0 -U -c 1000 10.2.1.1

# Distributed flood from several hosts in parallel
mininet> h1 arping -I h1-eth0 -c 1000 10.4.15.1 &
mininet> h2 arping -I h2-eth0 -c 1000 10.4.14.1 &
mininet> h3 arping -I h3-eth0 -c 1000 10.4.13.1 &

# MAC/IP spoofing test (install scapy inside a host if needed):
mininet> h1 python3 -c "from scapy.all import *; \
sendp(Ether(dst='ff:ff:ff:ff:ff:ff')/ARP(op=1, psrc='10.9.9.9', hwsrc='00:11:22:33:44:55', pdst='10.2.1.1'), iface='h1-eth0', count=500)"

--------------------------------------------------------------------------------
6. INSPECT THE OUTPUT
--------------------------------------------------------------------------------
# Live CSV
tail -f ~/Desktop/Arp_Flood_Detection-using-P4-Bmv2-main/dataset/arp_dataset_new.csv

# Number of rows collected
wc -l ~/Desktop/Arp_Flood_Detection-using-P4-Bmv2-main/dataset/arp_dataset_new.csv

# Rows by label (0 = normal, 1 = attack)
awk -F, 'NR>1 {c[$NF]++} END {for (k in c) print "label="k, c[k]}' \
  ~/Desktop/Arp_Flood_Detection-using-P4-Bmv2-main/dataset/arp_dataset_new.csv

# Controller log
tail -f /tmp/controller.log

--------------------------------------------------------------------------------
7. BLOCK / UNBLOCK (MANUAL, FROM ANY TERMINAL)
--------------------------------------------------------------------------------
# Install a block rule for one source IP on one switch (s1)
simple_switch_CLI --thrift-port 9090 <<< "table_add tbl_block_ip block_source 10.1.1.1 =>"

# Install a block rule for one source IP on all 5 switches
for P in 9090 9100 9110 9120 9190; do
  simple_switch_CLI --thrift-port $P <<< "table_add tbl_block_ip block_source 10.1.1.1 =>"
done

# Show current block rules on one switch
simple_switch_CLI --thrift-port 9090 <<< "table_dump tbl_block_ip"

# Remove ALL block rules on all 5 switches
for P in 9090 9100 9110 9120 9190; do
  simple_switch_CLI --thrift-port $P <<< "table_clear tbl_block_ip"
done
